```python
"""
BCC Flood Awareness FeatureServer → Supabase flood_zones
Sources:
  - Overall flood risk (high/med/low/very_low): flood_class field
  - Feb 2022 historic extent
"""
import os, sys, json, logging, requests
from supabase import create_client

logging.basicConfig(level=logging.INFO)
log = logging.getLogger(__name__)

SUPABASE = create_client(os.environ["SUPABASE_URL"], os.environ["SUPABASE_SERVICE_KEY"])

BCC_ENDPOINTS = [
    {
        "url": "https://services2.arcgis.com/dEKgZETqwmDAh1rP/ArcGIS/rest/services/Flood_Awareness_Flood_Risk_Overall/FeatureServer/0/query",
        "source": "bcc",
        "flood_class_field": "FLOOD_RISK",  # Adjust field name after inspecting response
    },
    {
        "url": "https://services2.arcgis.com/dEKgZETqwmDAh1rP/arcgis/rest/services/Flood_Awareness_Historic_Brisbane_River_and_Creek_Floods_Feb2022/FeatureServer/0/query",
        "source": "bcc",
        "flood_class_field": "FLOOD_TYPE",
    },
]

FLOOD_CLASS_MAP = {
    "HIGH": "high", "High": "high", "MEDIUM": "medium", "Medium": "medium",
    "LOW": "low", "Low": "low", "VERY LOW": "very_low", "Very Low": "very_low",
}

def fetch_paginated(url: str, page_size: int = 1000) -> list:
    features = []
    offset = 0
    while True:
        params = {
            "where": "1=1", "outFields": "*", "f": "geojson",
            "resultOffset": offset, "resultRecordCount": page_size,
        }
        resp = requests.get(url, params=params, timeout=60)
        resp.raise_for_status()
        data = resp.json()
        batch = data.get("features", [])
        features.extend(batch)
        if len(batch) < page_size:
            break
        offset += page_size
    return features

def geojson_to_multipolygon_wkt(geometry: dict) -> str | None:
    """Convert GeoJSON geometry to WKT MultiPolygon for PostGIS."""
    if geometry is None:
        return None
    gtype = geometry.get("type", "")
    coords = geometry.get("coordinates", [])
    if gtype == "Polygon":
        # Wrap single polygon in MultiPolygon
        rings = ", ".join(
            "(" + ", ".join(f"{x} {y}" for x, y in ring) + ")"
            for ring in coords
        )
        return f"MULTIPOLYGON(({rings}))"
    elif gtype == "MultiPolygon":
        polys = []
        for poly in coords:
            rings = ", ".join(
                "(" + ", ".join(f"{x} {y}" for x, y in ring) + ")"
                for ring in poly
            )
            polys.append(f"({rings})")
        return f"MULTIPOLYGON({', '.join(polys)})"
    return None

def ingest_endpoint(endpoint: dict) -> int:
    features = fetch_paginated(endpoint["url"])
    records = []
    for feat in features:
        props = feat.get("properties", {})
        geom = feat.get("geometry")
        wkt = geojson_to_multipolygon_wkt(geom)
        if not wkt:
            continue
        raw_class = props.get(endpoint["flood_class_field"], "")
        flood_class = FLOOD_CLASS_MAP.get(str(raw_class).strip(), raw_class.lower() if raw_class else None)
        council = props.get("COUNCIL", props.get("council", "BCC"))
        records.append({
            "source": endpoint["source"],
            "flood_class": flood_class,
            "council": council,
            "country_code": "AU",
            "region_slug": "brisbane",
            "geometry": f"SRID=4326;{wkt}",
        })
    if not records:
        log.warning(f"WARN: {endpoint['source']} — no records to upsert")
        return 0
    SUPABASE.table("flood_zones").upsert(records, on_conflict="source,flood_class,council").execute()
    return len(records)

def main():
    total = 0
    for ep in BCC_ENDPOINTS:
        try:
            n = ingest_endpoint(ep)
            log.info(f"OK: bcc ({ep['url'].split('/')[10]}) — {n} records upserted")
            total += n
        except Exception as e:
            log.warning(f"WARN: bcc failed: {e}")
    log.info(f"OK: bcc total — {total} records upserted")

if __name__ == "__main__":
    main()
```

"""
QLD Flood Extent 2011 (Grantham + Brisbane) → Supabase flood_zones
Source: https://www.data.qld.gov.au/dataset/flood-extent-series
Downloads GPKG, extracts Lockyer Valley / Grantham polygons.
"""
import os, sys, logging, tempfile, requests
import geopandas as gpd
from supabase import create_client

logging.basicConfig(level=logging.INFO)
log = logging.getLogger(__name__)

SUPABASE = create_client(os.environ["SUPABASE_URL"], os.environ["SUPABASE_SERVICE_KEY"])

# QLD Data portal — 2011 flood extent GPKG
# NOTE: Verify this URL against https://www.data.qld.gov.au/dataset/flood-extent-series
# before first run — the direct file URL may require navigating the dataset page.
QLD_2011_URL = "https://www.data.qld.gov.au/dataset/flood-extent-series/resource/b6c3ab5f-7a32-4afc-b706-9a4f1f42e2f7/download/qld-flood-extent-2011.gpkg"

def main():
    try:
        log.info("Downloading QLD 2011 flood extent GPKG...")
        with tempfile.NamedTemporaryFile(suffix=".gpkg", delete=False) as tmp:
            resp = requests.get(QLD_2011_URL, timeout=120, stream=True)
            resp.raise_for_status()
            for chunk in resp.iter_content(chunk_size=8192):
                tmp.write(chunk)
            tmp_path = tmp.name

        gdf = gpd.read_file(tmp_path)
        # Reproject to WGS84 if needed
        if gdf.crs and gdf.crs.to_epsg() != 4326:
            gdf = gdf.to_crs(epsg=4326)

        # Filter to Lockyer Valley / Grantham area
        # Grantham bbox: lon 152.1–152.3, lat -27.7 to -27.5
        grantham_mask = (
            (gdf.geometry.bounds["minx"] >= 152.0) &
            (gdf.geometry.bounds["maxx"] <= 152.4) &
            (gdf.geometry.bounds["miny"] >= -27.8) &
            (gdf.geometry.bounds["maxy"] <= -27.4)
        )
        grantham_gdf = gdf[grantham_mask]

        records = []
        for _, row in grantham_gdf.iterrows():
            geom = row.geometry
            if geom is None or geom.is_empty:
                continue
            # Ensure MultiPolygon
            if geom.geom_type == "Polygon":
                from shapely.geometry import MultiPolygon
                geom = MultiPolygon([geom])
            records.append({
                "source": "qld_2011",
                "flood_class": "high",
                "council": "Lockyer Valley Regional Council",
                "country_code": "AU",
                "region_slug": "grantham",
                "geometry": f"SRID=4326;{geom.wkt}",
                "data_date": "2011-01-10",
            })

        if not records:
            log.warning("WARN: qld_2011 — no Grantham polygons found in GPKG")
            return

        SUPABASE.table("flood_zones").upsert(records, on_conflict="source,flood_class,council").execute()
        log.info(f"OK: qld_2011 — {len(records)} records upserted")

    except Exception as e:
        log.warning(f"WARN: qld_2011 failed: {e}")
        sys.exit(1)

if __name__ == "__main__":
    main()

"""
QLD WMIP Real-Time Gauge API → Supabase flood_forecasts
Fetches live river levels for:
  - Brisbane River at City: approx -27.47, 153.02
  - Lockyer Creek at Grantham: approx -27.47, 152.33

WMIP API base: https://water-monitoring.information.qld.gov.au/
Docs: https://water-monitoring.information.qld.gov.au/wini/Documents/WMIP_API_2025.pdf

Station IDs must be discovered from the WMIP station search API.
The script auto-discovers stations near the target coordinates.
"""
import os, sys, logging, requests
from datetime import date, datetime
from supabase import create_client

logging.basicConfig(level=logging.INFO)
log = logging.getLogger(__name__)

SUPABASE = create_client(os.environ["SUPABASE_URL"], os.environ["SUPABASE_SERVICE_KEY"])

WMIP_BASE = "https://water-monitoring.information.qld.gov.au"

# Target gauges: (name, lat, lon, search_radius_km)
TARGET_GAUGES = [
    ("Brisbane River at City", -27.47, 153.02, 5.0),
    ("Lockyer Creek at Grantham", -27.47, 152.33, 10.0),
]

def find_station(lat: float, lon: float, radius_km: float) -> dict | None:
    """Search WMIP for nearest station to given coordinates."""
    try:
        # WMIP station search — spatial bounding box approach
        delta = radius_km / 111.0  # approx degrees
        url = f"{WMIP_BASE}/wini/monitoring/station"
        params = {
            "minLat": lat - delta, "maxLat": lat + delta,
            "minLon": lon - delta, "maxLon": lon + delta,
            "f": "json",
        }
        resp = requests.get(url, params=params, timeout=30)
        resp.raise_for_status()
        stations = resp.json()
        if isinstance(stations, list) and stations:
            # Return the closest station
            return min(
                stations,
                key=lambda s: abs(s.get("latitude", 0) - lat) + abs(s.get("longitude", 0) - lon)
            )
    except Exception as e:
        log.warning(f"WARN: wmip station search failed: {e}")
    return None

def fetch_gauge_reading(station_id: str) -> dict | None:
    """Fetch latest reading for a station."""
    try:
        url = f"{WMIP_BASE}/wini/monitoring/variable/{station_id}"
        resp = requests.get(url, params={"f": "json"}, timeout=30)
        resp.raise_for_status()
        return resp.json()
    except Exception as e:
        log.warning(f"WARN: wmip reading fetch failed for {station_id}: {e}")
    return None

def main():
    records = []
    today = date.today().isoformat()

    for name, lat, lon, radius in TARGET_GAUGES:
        try:
            station = find_station(lat, lon, radius)
            if not station:
                log.warning(f"WARN: wmip — no station found near {name} ({lat}, {lon})")
                continue

            station_id = station.get("stationId") or station.get("id")
            reading = fetch_gauge_reading(station_id)
            if not reading:
                continue

            # Extract discharge — field names vary; try common patterns
            discharge = None
            for field in ["value", "discharge", "streamflow", "level"]:
                if isinstance(reading, dict) and field in reading:
                    discharge = reading[field]
                    break
                elif isinstance(reading, list) and reading:
                    discharge = reading[0].get(field)
                    if discharge is not None:
                        break

            records.append({
                "source": "wmip",
